    23→    23→export function hitungUsiaBulan(tanggalLahir: string, tanggalAcuan = new Date()): number {
    24→    24→  const lahir = new Date(tanggalLahir);
    25→    25→  const diff = tanggalAcuan.getTime() - lahir.getTime();
    26→    26→  return Math.max(0, Math.floor(diff / (1000 * 60 * 60 * 24 * 30.4375)));
    27→    27→}

    29→    29→export function formatTanggalID(iso: string): string {
    30→    30→  const bulan = ['Jan', 'Feb', 'Mar', 'Apr', 'Mei', 'Jun', 'Jul', 'Agu', 'Sep', 'Okt', 'Nov', 'Des'];
    31→    31→  const d = new Date(iso);
    32→    32→  return `${d.getDate()} ${bulan[d.getMonth()]} ${d.getFullYear()}`;
    33→    33→}
