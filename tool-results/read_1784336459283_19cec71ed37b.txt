   369→// --- Today's Summary Widget ---
   370→function TodaysSummary() {
   371→  const pengukuranList = useStore((s) => s.pengukuranList);
   372→  const jadwalList = useStore((s) => s.jadwalList);
   373→  const notifikasiList = useStore((s) => s.notifikasiList);
   374→  const setView = useStore((s) => s.setView);
   375→
   376→  const todayStr = new Date().toISOString().slice(0, 10);
   377→  const today = new Date();
   378→
   379→  const todayPengukuran = pengukuranList.filter(
   380→    (p) => p.tanggalPengukuran === todayStr,
   381→  );
   382→  const todayJadwal = jadwalList.filter((j) => j.tanggal === todayStr);
   383→  const unreadNotif = notifikasiList.filter((n) => !n.dibaca).length;
   384→  const todayGiziBermasalah = todayPengukuran.filter(
   385→    (p) => p.statusGizi !== "Normal" && p.statusGizi !== "Risiko Gizi Lebih",
   386→  ).length;
   387→
   388→  const greeting =
   389→    today.getHours() < 11
   390→      ? "Selamat pagi"
   391→      : today.getHours() < 15
   392→        ? "Selamat siang"
   393→        : today.getHours() < 18
   394→          ? "Selamat sore"
   395→          : "Selamat malam";
   396→  const todayLabel = today.toLocaleDateString("id-ID", {
   397→    weekday: "long",
   398→    day: "numeric",
   399→    month: "long",
   400→    year: "numeric",
   401→  });
   402→
   403→  return (
   404→    <Card className="relative overflow-hidden border-emerald-200 bg-gradient-to-br from-emerald-50 via-teal-50 to-emerald-50 p-5 dark:border-emerald-500/20 dark:from-emerald-500/5 dark:via-teal-500/5 dark:to-emerald-500/5">
